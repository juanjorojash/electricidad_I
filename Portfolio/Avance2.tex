\begin{enumerate}
    \item Halle $i_1(t)$ e $i_2(t)$ para $t>0$ en el circuito de la figura.

    \begin{figure}[H]
        \centering
        \begin{circuitikz}
            % Cinco ramas conectadas entre los mismos nodos superior e inferior.
            \draw (0,0) to[I,l_=\SI{5}{\ampere}] (0,4) -- (10,4);
            \draw (0,0) -- (10,0);
            \draw (2.5,4) to[R,l=\SI{6}{\ohm}] (2.5,0);
            \draw (5,4) to[spst] (5,0);
            \node[left] at (4.8,2) {$t=0$};
            \draw (7.5,4) to[R,l_=\SI{5}{\ohm}] (7.5,2)
                            to[L,l_=\SI{2.5}{\henry}] (7.5,0);
            \draw (10,4) to[R,l=\SI{20}{\ohm}] (10,2)
                         to[L,l=\SI{4}{\henry}] (10,0);
            \draw[-{Latex[length=2mm]}] (7.85,3.95) -- (7.85,3.40)
                node[midway,right] {$i_1$};
            \draw[-{Latex[length=2mm]}] (10.35,3.95) -- (10.35,3.40)
                node[midway,right] {$i_2$};
        \end{circuitikz}
        \caption{Circuito para el cálculo de $i_1(t)$ e $i_2(t)$.}
        \label{fig:avance2-circuito}
    \end{figure}

    \textbf{Respuesta:}
    \[
    \begin{aligned}
        i_1(t) &= 2{,}4e^{-2t}\,\mathrm{A},\\
        i_2(t) &= 0{,}6e^{-5t}\,\mathrm{A}.
    \end{aligned}
    \]

    \item El capacitor de la figura se encuentra cargado con $v_c(0)=\SI{8}{\volt}$. Escriba las expresiones matemáticas para $v_c$ e $i_x$ para todo tiempo $t>0$.

    \begin{figure}[H]
        \centering
        \begin{circuitikz}
            \draw (0,0) node[ocirc]{} node[left]{$+\SI{10}{\volt}$}
                to[R,l=\SI{2}{\kilo\ohm}] (3,0)
                -- (4,0) to[R,l=\SI{6.8}{\kilo\ohm}] (6.5,0)
                to[spst] (8,0)
                to[C,l=\SI{30}{\micro\farad}] (10.5,0)
                -- (11.4,0) node[ocirc]{} node[right]{$+\SI{15}{\volt}$};
            \draw (3,0) node[circ]{} -- (3,-0.7) node[ground]{};
            \draw[-{Latex[length=2mm]}] (3.95,0.30) -- (3.35,0.30)
                node[midway,above] {$i_x$};
            \node[above] at (7.25,0.35) {$t=0{,}4$};
            \node at (8.75,-0.55) {$+$};
            \node at (9.75,-0.55) {$-$};
            \node at (9.25,-0.85) {$v_c$};
        \end{circuitikz}
        \caption{Circuito para el cálculo de $v_c$ e $i_x$.}
        \label{fig:avance2-capacitor}
    \end{figure}

    \textbf{Respuesta:}
    \[
    v_c(t)=
    \begin{cases}
        8\,\mathrm{V}, & 0<t<0{,}4,\\
        (163e^{-4{,}96t}-15)\,\mathrm{V}, & t\ge 0{,}4.
    \end{cases}
    \]
    \[
    i_x(t)=
    \begin{cases}
        0\,\mathrm{mA}, & 0<t<0{,}4,\\
        24e^{-4{,}96t}\,\mathrm{mA}, & t\ge 0{,}4.
    \end{cases}
    \]

    \item En el circuito RLC en paralelo de la figura, halle $v(t)$ para $t>0$.

    \begin{figure}[H]
        \centering
        \begin{circuitikz}
            \draw (0,0) to[I,l_=\SI{4.5}{\ampere}] (0,4)

    \textbf{Respuesta:}
    \[
        v(t)=\bigl(-150e^{-2{,}5t}+150e^{-10t}\bigr)\,\mathrm{V},
        \qquad t>0.
    \]

    \item Halle la corriente $i(t)$ para $t<0$ y $t>0$ en el circuito de la figura.

    \begin{figure}[H]
        \centering
        \begin{circuitikz}
            \draw (0,0) to[V,l_=\SI{40}{\volt}] (0,3.6)

            \node[above] at (3.2,4) {$t=0$};

    \[
        i(t)=6\,\mathrm{A}.
    \]

    \[
        i(t)=6e^{-4t}\,\mathrm{A}.
    \]

    \item Sea $v_C(0)=\SI{60}{\volt}$. Determine $v_C(t)$, $v_x(t)$ e $i_o(t)$ para $t\geq 0$ en el circuito de la figura.

    \begin{figure}[H]
        \centering
        \begin{circuitikz}
            \draw (0,0) to[R,l_=\SI{12}{\ohm}] (0,3.5)

    \[
        v_C(t)=60e^{-0{,}25t}\,\mathrm{V},\qquad t\geq 0.
    \]

    \textbf{Respuesta para $v_x$:}
    \[
        v_x(t)=20e^{-0{,}25t}\,\mathrm{V},\qquad t\geq 0.
    \]

    \textbf{Respuesta para $i_o$:}
    \[
        i_o(t)=-5e^{-0{,}25t}\,\mathrm{A},\qquad t\geq 0.
    \]
