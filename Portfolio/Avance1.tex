\newcommand{\fuentehorizontal}[7]{%
  % #1: x izquierda, #2: y, #3: x derecha, #4: etiqueta, #5: signo izq., #6: signo der., #7: posición de etiqueta
  \draw (#1,#2) -- ($ (#1,#2)!0.43!(#3,#2) $);
  \draw ($ (#1,#2)!0.57!(#3,#2) $) -- (#3,#2);
  \path let \p1=($ (#1,#2)!0.5!(#3,#2) $) in coordinate (src) at (\x1,\y1);
  \draw (src) circle (0.34);
  \node at ($ (src)+(-0.14,0) $) {#5};
  \node at ($ (src)+(0.14,0) $) {#6};
  \node[#7] at (src) {#4};
}

\begin{enumerate}
    \item Utilizando solamente LVK y LCK obtenga $V_x$ e $i_x$. Indique si la fuente $i_x$ consume o entrega potencia.

    \begin{center}
    \begin{circuitikz}[scale=0.85,transform shape]
        \draw (0,0) -- (7,0);
        \draw (0,0) to[V,l=\SI{4}{\volt}] (0,2);
        \draw (0,2) to[R,l=\SI{8}{\ohm},i>^=\SI{200}{\milli\ampere}] (2.4,2)
              to[R,l=\SI{4}{\ohm}] (4.8,2) -- (7,2);
        \draw (2.4,2) to[R,l=\SI{2}{\ohm}] (2.4,0);
        \draw (4.8,2) to[R,l=\SI{3}{\ohm}] (4.8,0);
        \draw (7,0) to[I] (7,2);
        \node[above] at (7.35,1.50) {$i_x$};
        \node[right] at (7.80,1.68) {$+$};
        \node[right] at (7.80,1.00) {$V_x$};
        \node[right] at (7.80,0.25) {$-$};
    \end{circuitikz}
    \end{center}

    \vspace{0.35cm}
    \noindent\hspace*{-\leftmargin}Respuesta:\\
    \noindent\hspace*{-\leftmargin}\textbf{$V_x=6{,}4\,\mathrm{V}$, $i_x=3{,}133\,\mathrm{A}$}
    \vspace{0.25cm}

    \item Para el circuito de la siguiente figura, plantee la ecuación de la supermalla y encuentre los valores de $i_1$, $i_2$, $i_3$, $i_4$, $i_x$.
